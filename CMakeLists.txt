cmake_minimum_required(VERSION 3.16)
project(BitWeave VERSION 1.4.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(bitweave src/bitweave.c)
add_executable(bitpack src/bitpack.c)
foreach(t bitweave bitpack)
  target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
enable_testing()
add_test(NAME version COMMAND bitweave --version)
